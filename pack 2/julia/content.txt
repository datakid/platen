@@C custom-iterators | custom iterators | 3 | Makes a type iterable by defining Base.iterate.
@@E
struct Squares
    n::Int
end
Base.iterate(s::Squares, i=1) = i > s.n ? nothing : (i^2, i + 1)
Base.length(s::Squares) = s.n

@@N nothing ==> signals the end
@@N (i^2, i + 1) ==> value and next state
@@E
struct Countdown
    from::Int
end
function Base.iterate(c::Countdown, state=c.from)
    state < 1 && return nothing
    return (state, state - 1)
end
for n in Countdown(3)
    println(n)
end
@@E
struct Evens
    limit::Int
end
function Base.iterate(e::Evens, n=0)
    n > e.limit ? nothing : (n, n + 2)
end
Base.length(e::Evens) = div(e.limit, 2) + 1
println(collect(Evens(6)))
